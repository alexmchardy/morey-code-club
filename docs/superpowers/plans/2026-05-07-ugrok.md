# ugrok Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-paced code stepper and quiz tool that teaches Python control flow through JSON-defined groksets, with a fully implemented conditionals grokset.

**Architecture:** Single-page HTML app (`ugrok/index.html`) with a separate CSS file. Groksets are JSON files fetched at runtime. Skulpt runs student code for quiz validation. Pre-authored execution traces power the code stepper. Cerebras API provides LLM-powered help.

**Tech Stack:** HTML/CSS/JS (no build tools), CodeMirror 5.65.16, Skulpt 1.2.0, Cerebras API (llama3.1-8b), Web Audio API

    // --- Sound Effects ---
    function createSounds() {
      let ctx;

      function getCtx() {
        if (!ctx) ctx = new (window.AudioContext || window.webkitAudioContext)();
        return ctx;
      }

      function playTone(freq, duration, type = 'square', volume = 0.15) {
        try {
          const ac = getCtx();
          const osc = ac.createOscillator();
          const gain = ac.createGain();
          osc.type = type;
          osc.frequency.setValueAtTime(freq, ac.currentTime);
          gain.gain.setValueAtTime(volume, ac.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, ac.currentTime + duration);
          osc.connect(gain);
          gain.connect(ac.destination);
          osc.start(ac.currentTime);
          osc.stop(ac.currentTime + duration);
        } catch (e) {}
      }

      function playNotes(notes, type = 'square', volume = 0.15) {
        try {
          const ac = getCtx();
          let time = ac.currentTime;
          for (const [freq, dur] of notes) {
            const osc = ac.createOscillator();
            const gain = ac.createGain();
            osc.type = type;
            osc.frequency.setValueAtTime(freq, time);
            gain.gain.setValueAtTime(volume, time);
            gain.gain.exponentialRampToValueAtTime(0.001, time + dur);
            osc.connect(gain);
            gain.connect(ac.destination);
            osc.start(time);
            osc.stop(time + dur);
            time += dur;
          }
        } catch (e) {}
      }

      return {
        step()      { playTone(440, 0.06, 'sine', 0.05); },
        correct()   { playNotes([[523, 0.08], [659, 0.08], [784, 0.12]], 'square', 0.12); },
        wrong()     { playNotes([[300, 0.1], [200, 0.15]], 'sawtooth', 0.1); },
        complete()  { playNotes([[523, 0.1], [659, 0.1], [784, 0.1], [1047, 0.2]], 'square', 0.15); },
      };
    }

    const sounds = createSounds();

            if (ln === currentLine) {

          stepIndex++;
          sounds.step();
          updateDisplay();
          if (stepIndex >= trace.length - 1) stopAutoPlay();

        if (stepIndex > -1) {

          updateDisplay();

        updateDisplay();

    function builtinRead(filename) {
      if (Sk.builtinFiles === undefined || Sk.builtinFiles["files"][filename] === undefined) {
        throw "File not found: '" + filename + "'";
      }
      return Sk.builtinFiles["files"][filename];
    }

    async function runPython(code) {
      const output = [];
      Sk.configure({
        output: (text) => output.push(text),
        read: builtinRead,
        __future__: Sk.python3,
      });
      Sk.execLimit = 5000;

      try {
        const module = await Sk.misceval.asyncToPromise(() => {
          return Sk.importMainWithBody("<stdin>", false, code, true);
        });

        // Extract variable values from module
        const state = {};
        for (const key of Object.keys(module.$d)) {
          if (key.startsWith('__')) continue;
          try {
            state[key] = Sk.ffi.remapToJs(module.$d[key]);
          } catch {
            state[key] = String(module.$d[key]);
          }
        }

        return { success: true, output, state, error: null };
      } catch (err) {
        return { success: false, output, state: {}, error: err.toString() };
      }
    }

      });

      // Save code on change

      document.getElementById('quizRunBtn').addEventListener('click', async () => {

          }
        } else if (validation.type === 'state') {

          for (const [varName, expected] of Object.entries(validation.check)) {
            if (result.state[varName] !== expected) {
              passed = false;

**Files:**

- [ ] **Step 1: Implement Help button toggle and LLM request**

Add before the `// --- Init ---` section:

```javascript
    // --- Help System ---
    let helpCooldownTimer = null;

    function toggleHelp() {
      const panel = document.getElementById('helpPanel');
      panel.classList.toggle('open');
      if (panel.classList.contains('open')) {
        document.getElementById('helpInput').focus();
      }
    }

    async function askHelp() {
      const input = document.getElementById('helpInput');
      const question = input.value.trim();
      if (!question) return;

      const responseEl = document.getElementById('helpResponse');
      const askBtn = document.getElementById('helpAskBtn');
      responseEl.textContent = 'Thinking...';
      askBtn.disabled = true;

      const exercise = currentGrokset.exercises[currentExerciseIndex];
      const systemPrompt = buildHelpPrompt(exercise);

      try {
        const response = await fetch('https://api.cerebras.ai/v1/chat/completions', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${CEREBRAS_API_KEY}`,
          },
          body: JSON.stringify({
            model: 'llama3.1-8b',
            messages: [
              { role: 'system', content: systemPrompt },
              { role: 'user', content: question },
            ],
            max_tokens: 200,
          }),
        });

        const data = await response.json();
        responseEl.textContent = data.choices?.[0]?.message?.content || 'No response';
      } catch (err) {
        responseEl.textContent = 'Error: could not reach the help service.';
      }

      input.value = '';
      startHelpCooldown();
    }

    function buildHelpPrompt(exercise) {
      const topic = currentGrokset.title;

      if (exercise.type === 'explanation' || exercise.type === 'stepper') {
        return `You are a friendly coding tutor helping an elementary/middle school student understand Python. The student is learning about ${topic}. They are currently looking at this code:\n${exercise.code}\nAnswer their question simply and concisely. Use analogies when helpful. Keep responses under 3 sentences.`;
      }

      let prompt = `You are a friendly coding tutor. The student is working on this quiz question:\n${exercise.question}`;
      if (exercise.code) prompt += `\n\nCode:\n${exercise.code}`;
      prompt += `\n\nGive a helpful hint that guides them toward the answer WITHOUT giving it away. Use leading questions or point out what to focus on. Keep responses under 3 sentences.`;

      if (exercise.type === 'quiz-code' && quizEditor) {
        const studentCode = quizEditor.getValue();
        if (studentCode && studentCode !== exercise.starterCode) {
          prompt += `\n\nThe student's current code:\n${studentCode}`;
        }
      }

      return prompt;
    }

    function startHelpCooldown() {
      const askBtn = document.getElementById('helpAskBtn');
      const cooldownEl = document.getElementById('helpCooldown');
      askBtn.disabled = true;
      let remaining = 10;
      cooldownEl.textContent = `Wait ${remaining}s...`;
      helpCooldownTimer = setInterval(() => {
        remaining--;
        if (remaining <= 0) {
          clearInterval(helpCooldownTimer);
          helpCooldownTimer = null;
          askBtn.disabled = false;
          cooldownEl.textContent = '';
        } else {
          cooldownEl.textContent = `Wait ${remaining}s...`;
        }
      }, 1000);
    }

    document.getElementById('helpBtn').addEventListener('click', toggleHelp);
    document.getElementById('helpAskBtn').addEventListener('click', askHelp);
    document.getElementById('helpInput').addEventListener('keydown', (e) => {
      if (e.key === 'Enter' && !e.shiftKey) {
        e.preventDefault();
        askHelp();
      }
    });
```
